# Módulo de Clientes — Implementation Plan

> **For agentic workers:** Use superpowers:subagent-driven-development or superpowers:executing-plans to implement task-by-task.

**Goal:** Add a complete Clientes module with customer registration, orders/cart, installments, digital signature, and dashboard.

**Architecture:** Sequential layers — DB migration first, then routes/menu, then each UI layer. Orders trigger stock movements via DB trigger. Soft-delete for customers with orders.

**Tech Stack:** TanStack Router (file-based), React Query, Supabase, shadcn/ui, Tailwind CSS 4, signature_pad, jsPDF

CREATE OR REPLACE FUNCTION public.tg_generate_order_number()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
DECLARE yr TEXT := to_char(now(),'YYYY'); seq INT;
BEGIN
  SELECT COALESCE(MAX(CAST(split_part(order_number,'-',3) AS INT)),0)+1
    INTO seq FROM public.orders
    WHERE organization_id=NEW.organization_id AND order_number LIKE 'ORD-'||yr||'-%';
  NEW.order_number := 'ORD-'||yr||'-'||lpad(seq::TEXT,4,'0');
  RETURN NEW;
END $$;

  FOR EACH ROW EXECUTE FUNCTION public.tg_generate_order_number();

CREATE OR REPLACE FUNCTION public.tg_order_stock_sync()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN
  IF NEW.status='confirmado' AND OLD.status!='confirmado' THEN
    INSERT INTO public.stock_movements(organization_id,product_id,movement_type,quantity,unit_cost,reason,reference,created_by)
    SELECT NEW.organization_id,oi.product_id,'saida',oi.quantity,oi.unit_price,'Venda '||NEW.order_number,NEW.order_number,NEW.seller_id
    FROM public.order_items oi WHERE oi.order_id=NEW.id AND oi.status='ativo';
  END IF;
  IF NEW.status='cancelado' AND OLD.status='confirmado' THEN
    INSERT INTO public.stock_movements(organization_id,product_id,movement_type,quantity,unit_cost,reason,reference,created_by)
    SELECT NEW.organization_id,oi.product_id,'entrada',oi.quantity,oi.unit_price,'Estorno '||NEW.order_number,NEW.order_number,NEW.seller_id
    FROM public.order_items oi WHERE oi.order_id=NEW.id AND oi.status='ativo';
  END IF;
  RETURN NEW;
END $$;

function ClientesPage() {
  const { data: profile } = useProfile();
  const orgId = profile?.active_org_id;
  const [search, setSearch] = useState("");
  const [statusFilter, setStatusFilter] = useState("todos");

  const { data: customers = [], isLoading } = useQuery({
    queryKey: ["customers", orgId],
    enabled: !!orgId,
    queryFn: async () => {
      const { data, error } = await supabase
        .from("customers")
        .select(`
          *,
          customer_addresses(city, state),
          orders(id, total_amount, payment_status, created_at, status)
        `)
        .eq("organization_id", orgId!)
        .is("deleted_at", null)
        .order("name");
      if (error) throw error;
      return data ?? [];
    },
  });

  // KPI calculations
  const total = customers.length;
  const ativos = customers.filter((c: any) => c.status === "ativo").length;
  const inadimplentes = customers.filter((c: any) => c.status === "inadimplente").length;
  const thisMonth = new Date(); thisMonth.setDate(1);
  const novosNoMes = customers.filter((c: any) => new Date(c.created_at) >= thisMonth).length;

  const filtered = customers.filter((c: any) => {
    const term = search.toLowerCase();
    const matchSearch = !search ||
      c.name?.toLowerCase().includes(term) ||
      c.cpf_cnpj?.includes(term) ||
      c.customer_addresses?.[0]?.city?.toLowerCase().includes(term);
    const matchStatus = statusFilter === "todos" || c.status === statusFilter;
    return matchSearch && matchStatus;
  });

  function getCity(c: any) {
    const addr = c.customer_addresses?.[0];
    if (!addr) return "—";
    return addr.city ? `${addr.city}/${addr.state}` : "—";
  }

  function getOrderStats(c: any) {
    const orders = (c.orders ?? []).filter((o: any) => o.status !== "cancelado");
    const total = orders.reduce((s: number, o: any) => s + Number(o.total_amount), 0);
    const pending = orders.filter((o: any) => o.payment_status === "pendente" || o.payment_status === "vencido")
      .reduce((s: number, o: any) => s + Number(o.total_amount), 0);
    const last = orders.sort((a: any, b: any) => new Date(b.created_at).getTime() - new Date(a.created_at).getTime())[0];
    return { count: orders.length, total, pending, lastDate: last?.created_at };
  }

  const brl = (n: number) => n.toLocaleString("pt-BR", { style: "currency", currency: "BRL" });

        latitude: lat,
